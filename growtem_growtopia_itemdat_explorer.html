                            1. Cloudflare Worker Script (Gratis 24/7 di Cloud)
                        

                        
                    

                    
export default {
  async fetch(request) {
    const corsHeaders = {
      "Access-Control-Allow-Origin": "*",
      "Access-Control-Allow-Methods": "GET, HEAD, OPTIONS",
      "Access-Control-Allow-Headers": "Content-Type",
      "Access-Control-Expose-Headers": "X-GT-Meta, X-GT-Items-URL, Last-Modified",
    };

    if (request.method === "OPTIONS") return new Response(null, { headers: corsHeaders });

    try {
      const endpoints = [
        { url: "https://www.growtopia1.com/growtopia/server_data.php", method: "POST" },
        { url: "https://www.growtopia2.com/growtopia/server_data.php", method: "POST" }
      ];

      let serverDataText = "";
      for (const ep of endpoints) {
        try {
          const res = await fetch(ep.url, {
            method: ep.method,
            headers: {
              "User-Agent": "UbiServices_SDK_2022.Release.9_PC64_ansi_static",
              "Content-Type": "application/x-www-form-urlencoded",
            },
            body: "version=5.62&platform=0&protocol=215",
          });
          const txt = await res.text();
          if (txt.includes("meta|")) {
            serverDataText = txt;
            break;
          }
        } catch (e) {}
      }

      const metaMatch = serverDataText.match(/meta\|([^\r\n]+)/);
      const cdnMatch = serverDataText.match(/cdnurl\|([^\r\n]+)/);

      if (!metaMatch) {
        throw new Error("IP Cloudflare diblokir WAF Ubisoft. Gunakan Node.js Proxy di PC (Opsi 2) agar memakai IP rumah Anda!");
      }

      const meta = metaMatch[1].trim();
      const cdnBase = cdnMatch ? `https://${cdnMatch[1].trim()}` : "https://ubistatic-a.akamaihd.net/0098";
      const itemsDatUrl = `${cdnBase}/${meta}/cache/items.dat`;

      const itemsRes = await fetch(itemsDatUrl, {
        headers: { "User-Agent": "UbiServices_SDK_2022.Release.9_PC64_ansi_static" },
      });

      if (!itemsRes.ok) throw new Error("Gagal mengambil items.dat dari CDN");

      const headers = new Headers(corsHeaders);
      headers.set("Content-Type", "application/octet-stream");
      headers.set("X-GT-Meta", meta);
      headers.set("X-GT-Items-URL", itemsDatUrl);

      return new Response(itemsRes.body, { status: 200, headers });
    } catch (err) {
      return new Response(JSON.stringify({ error: err.message }), {
        status: 500,
        headers: { ...corsHeaders, "Content-Type": "application/json" },
      });
    }
  },
};

                


                
                

                    

                        

                            2. Node.js Express Server (Localhost / VPS)
                        

                        
                    

                    
const express = require("express");
const cors = require("cors");
const fs = require("fs");
const path = require("path");
const app = express();

app.use(cors({ exposedHeaders: ["X-GT-Meta", "X-GT-Items-URL"] }));

app.get(["/", "/items.dat"], async (req, res) => {
  try {
    const hosts = [
      "https://www.growtopia1.com/growtopia/server_data.php",
      "https://www.growtopia2.com/growtopia/server_data.php"
    ];

    let meta = null;
    let cdnBase = "https://ubistatic-a.akamaihd.net/0098";

    for (const url of hosts) {
      try {
        const serverDataRes = await fetch(url, {
          method: "POST",
          headers: {
            "User-Agent": "UbiServices_SDK_2022.Release.9_PC64_ansi_static",
            "Content-Type": "application/x-www-form-urlencoded",
          },
          body: "version=5.90&platform=0&protocol=225",
        });
        const txt = await serverDataRes.text();
        const mMatch = txt.match(/meta\|([^\r\n]+)/);
        const cMatch = txt.match(/cdnurl\|([^\r\n]+)/);
        if (mMatch) {
          meta = mMatch[1].trim();
          if (cMatch) cdnBase = `https://${cMatch[1].trim()}`;
          break;
        }
      } catch (e) {}
    }

    if (meta) {
      const itemsDatUrl = `${cdnBase}/${meta}/cache/items.dat`;
      const itemsRes = await fetch(itemsDatUrl, {
        headers: { "User-Agent": "UbiServices_SDK_2022.Release.9_PC64_ansi_static" }
      });
      if (itemsRes.ok) {
        const buffer = Buffer.from(await itemsRes.arrayBuffer());
        res.set({
          "Content-Type": "application/octet-stream",
          "X-GT-Meta": meta,
          "X-GT-Items-URL": itemsDatUrl,
        });
        return res.send(buffer);
      }
    }

    // Fallback otomatis ke cache lokal Growtopia jika server_data.php sedang maintenance
    const localCache = path.join(process.env.LOCALAPPDATA || "", "Growtopia", "cache", "items.dat");
    if (fs.existsSync(localCache)) {
      const buffer = fs.readFileSync(localCache);
      res.set({
        "Content-Type": "application/octet-stream",
        "X-GT-Meta": "Local-PC-Cache",
      });
      return res.send(buffer);
    }

    throw new Error("Server Growtopia tidak merespons parameter meta dan cache lokal tidak ditemukan.");
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

app.listen(3000, () => console.log("Proxy berjalan di http://localhost:3000/items.dat"));
